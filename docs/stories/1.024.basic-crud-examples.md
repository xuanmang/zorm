### 错误处理模式

[Source: docs/functional_spec.md - 2.1.3 错误处理系统]

```zig
// 1. 传播错误
pub fn findUser(db: *DB, id: i64) !User {
    var query = try db.newSelect(User);
    defer query.deinit();
    
    try query.where("id = ?", .{id});
    return try query.scanOne();
}

// 2. 捕获并处理
pub fn findUserSafe(db: *DB, id: i64) ?User {
    return findUser(db, id) catch |err| {
        std.log.err("Failed to find user: {}", .{err});
        return null;
    };
}

// 3. 错误类型判断
pub fn handleQueryError(db: *DB, id: i64) !User {
    return findUser(db, id) catch |err| switch (err) {
        error.NoRows => {
            std.log.warn("User {} not found", .{id});
            return error.NoRows;
        },
        error.ConnectionFailed => {
            std.log.err("Database connection failed", .{});
            return error.ConnectionFailed;
        },
        else => return err,
    };
}
```

### 内存管理要点

[Source: docs/architecture.md - 内存管理策略]

```zig
// ✅ 正确的内存管理
pub fn loadUsers(allocator: Allocator, db: *DB) !std.ArrayList(User) {
    var users: std.ArrayList(User) = .{};  // Zig 0.15.2+
    errdefer users.deinit(allocator);       // 错误时自动清理
    
    var query = try db.newSelect(User);
    defer query.deinit();
    
    try query.scan(&users);
    return users;  // 调用者负责 deinit
}

// 使用
pub fn example(allocator: Allocator, db: *DB) !void {
    var users = try loadUsers(allocator, db);
    defer users.deinit(allocator);  // 清晰的清理
    
    // 使用 users
}
```

### 代码风格约定

[Source: docs/architecture.md - 核心设计原则]

1. **显式优于隐式**: 所有 allocator 显式传递
2. **使用 defer 清理**: 确保资源释放
3. **错误必须处理**: 使用 `!T` 和 `try`/`catch`
4. **注释说明 why**: 解释为什么这样实现,而非 what

### 安全实践

[Source: docs/prd-examples.md - NFR8: 安全实践]

1. **参数化查询**: 始终使用 `?` 占位符,防止 SQL 注入
   ```zig
   // ✅ 正确
   try query.where("name = ?", .{user_input});
   
   // ❌ 错误 - SQL 注入风险
   const sql = try std.fmt.allocPrint(allocator, "WHERE name = '{s}'", .{user_input});
   ```

2. **UPDATE/DELETE 必须有 WHERE**: 防止意外修改所有数据
   ```zig
   // ✅ 正确
   try update.where("id = ?", .{123});
   
   // ❌ 危险 - 会更新所有行
   try update.set("status = 'active'").exec();
   ```

3. **使用事务保护关键操作**: 确保数据一致性
   ```zig
   const tx = try db.beginTx();
   errdefer tx.rollback() catch {};
   
   // 执行多个操作
   
   try tx.commit();
   ```

```zig
const std = @import("std");
const testing = std.testing;
const zorm = @import("zorm");

test "SELECT query builder" {
    const allocator = testing.allocator;
    
    const db = try zorm.DB.initMock(allocator, .postgresql);
    defer db.deinit();
    
    var query = try db.newSelect(User);
    defer query.deinit();
    
    try query.where("age > ?", .{18});
    const sql = try query.buildSQL();
    defer allocator.free(sql);
    
    try testing.expectEqualStrings(
        "SELECT * FROM users WHERE age > $1",
        sql,
    );
}

test "INSERT with values" {
    const allocator = testing.allocator;
    
    const db = try zorm.DB.initMock(allocator, .postgresql);
    defer db.deinit();
    
    const user = User{
        .name = "Test",
        .email = "test@example.com",
        .age = 25,
        .created_at = 0,
        .updated_at = 0,
    };
    
    var insert = try db.newInsert(User);
    defer insert.deinit();
    
    try insert.value(user);
    const sql = try insert.buildSQL();
    defer allocator.free(sql);
    
    // 验证生成的 SQL
    try testing.expect(std.mem.containsAtLeast(u8, sql, 1, "INSERT INTO users"));
}
```
