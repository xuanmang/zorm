### ZORM API 引用

#### 查询构建器高级 API
[Source: docs/functional_spec.md - 2.2.1 SELECT 查询]

```zig
// WHERE 组合
var query = try db.newSelect(User);
try query
    .where("age > ?", .{18})
    .where("status = ?", .{"active"})  // AND
    .whereOr("role = ?", .{"admin"});  // OR

// JOIN
try query
    .leftJoin("posts", "posts.user_id = users.id")
    .leftJoin("comments", "comments.post_id = posts.id");

// 聚合
try query
    .column("users.id")
    .column("COUNT(posts.id) as post_count")
    .groupBy("users.id")
    .having("COUNT(posts.id) > ?", .{5});

// 排序和分页
try query
    .orderBy("created_at", .desc)
    .limit(10)
    .offset(20);
```

#### 事务管理 API
[Source: docs/functional_spec.md - 推断的事务 API]

```zig
// 基础事务
const tx = try db.beginTx();
errdefer tx.rollback() catch {};

// 在事务中执行操作
var update = try tx.newUpdate(Account);
try update.set("balance = balance - ?", .{100}).exec();

// 提交
try tx.commit();

// Savepoint
try tx.savepoint("sp1");
// ... 操作
try tx.rollbackTo("sp1");  // 回滚到 savepoint
// 或
try tx.releaseSavepoint("sp1");  // 释放 savepoint

// 只读事务
const tx = try db.beginTx(.{ .read_only = true });

// 隔离级别
const tx = try db.beginTx(.{ .isolation_level = .serializable });
```

### 查询构建器设计哲学

[Source: docs/architecture.md - QueryBuilder 设计]

ZORM 的查询构建器遵循以下原则:

1. **类型安全**: 编译时验证查询的有效性
2. **链式 API**: 流畅的接口,易于阅读和编写
3. **SQL 透明**: 生成的 SQL 可预测,无隐藏查询
4. **零成本抽象**: 编译后性能接近手写 SQL

---

### 事务最佳实践

[Source: 数据库最佳实践]

1. **保持事务简短**: 减少锁持有时间
2. **使用 errdefer**: 确保错误时自动回滚
3. **选择合适的隔离级别**: 平衡一致性和性能
4. **避免长时间运行的事务**: 防止死锁
5. **使用 Savepoint 实现部分回滚**: 提高灵活性

### Testing

#### 测试示例

```zig
test "query builder - WHERE combination" {
    const allocator = testing.allocator;
    const db = try zorm.DB.initMock(allocator, .postgresql);
    defer db.deinit();
    
    var query = try db.newSelect(User);
    defer query.deinit();
    
    try query
        .where("age > ?", .{18})
        .where("status = ?", .{"active"})
        .whereOr("role = ?", .{"admin"});
    
    const sql = try query.buildSQL();
    defer allocator.free(sql);
    
    try testing.expect(std.mem.containsAtLeast(u8, sql, 1, "WHERE age > $1"));
    try testing.expect(std.mem.containsAtLeast(u8, sql, 1, "AND status = $2"));
    try testing.expect(std.mem.containsAtLeast(u8, sql, 1, "OR role = $3"));
}

test "transaction - commit" {
    const allocator = testing.allocator;
    const db = try zorm.DB.initMock(allocator, .postgresql);
    defer db.deinit();
    
    const tx = try db.beginTx();
    errdefer tx.rollback() catch {};
    
    // 执行操作
    var update = try tx.newUpdate(Account);
    defer update.deinit();
    
    try update.set("balance = balance - ?", .{100}).exec();
    
    // 提交
    try tx.commit();
    
    // 验证事务已提交
    try testing.expect(tx.isCommitted());
}
```
