      fn up_20250118_003_split_full_name(db: *DB) !void {
          // 1. 添加新列

            \\UPDATE users 
            \\SET first_name = SPLIT_PART(full_name, ' ', 1),
            \\    last_name = SPLIT_PART(full_name, ' ', 2)
            \\WHERE full_name IS NOT NULL;
          try db.newRaw(sql, .{}).exec();
          
          // 3. 删除旧列

      }

  - [ ] **示例 15: 批量数据迁移 (大表)**
    - [ ] 分批处理,避免锁表:
      ```zig
      fn up_20250118_005_migrate_large_table(db: *DB) !void {
          const batch_size = 1000;
          var offset: usize = 0;
          
          while (true) {
              const sql = 
                \\UPDATE old_table 
                \\SET new_column = process(old_column)
                \\WHERE id IN (
                \\  SELECT id FROM old_table 
                \\  WHERE new_column IS NULL 
                \\  LIMIT ?
                \\);
              const result = try db.newRaw(sql, .{batch_size}).exec();
              
              if (result.rows_affected == 0) break;
              offset += batch_size;
          }
      }
      ```

### Schema 迁移最佳实践

[Source: 数据库迁移最佳实践]

1. **版本号命名**: 使用时间戳 `YYYYMMDD_NNN_description`
2. **可逆迁移**: 每个迁移必须有 `up` 和 `down`
3. **小步迭代**: 每个迁移只做一件事
4. **测试迁移**: 在开发环境测试 up 和 down
5. **数据备份**: 生产环境迁移前备份数据
6. **避免破坏性变更**: 优先使用 ADD COLUMN 而非 ALTER COLUMN TYPE
7. **分批处理大表**: 避免长时间锁表

### 迁移系统架构

```zig
/// 迁移管理器
pub const Migrator = struct {
    db: *DB,
    allocator: Allocator,
    migrations: []const MigrationDef,
    
    pub fn init(allocator: Allocator, db: *DB, migrations: []const MigrationDef) Migrator {
        return .{
            .db = db,
            .allocator = allocator,
            .migrations = migrations,
        };
    }
    
    /// 运行所有待执行的迁移
    pub fn runMigrations(self: *Migrator) !void {
        // 确保迁移历史表存在
        try self.ensureMigrationTable();
        
        // 获取已应用的迁移
        const applied = try self.getAppliedMigrations();
        defer applied.deinit(self.allocator);
        
        // 运行新迁移
        for (self.migrations) |migration| {
            if (!applied.contains(migration.version)) {
                std.log.info("Running migration: {s}", .{migration.name});
                
                const tx = try self.db.beginTx();
                errdefer tx.rollback() catch {};
                
                try migration.up(tx);
                try self.recordMigration(tx, migration);
                try tx.commit();
                
                std.log.info("Migration completed: {s}", .{migration.name});
            }
        }
    }
    
    /// 回滚最后一个迁移
    pub fn rollback(self: *Migrator) !void {
        const last = try self.getLastMigration();
        if (last == null) {
            std.log.warn("No migrations to rollback", .{});
            return;
        }
        
        // 找到对应的迁移定义
        for (self.migrations) |migration| {
            if (std.mem.eql(u8, migration.version, last.?.version)) {
                std.log.info("Rolling back migration: {s}", .{migration.name});
                
                const tx = try self.db.beginTx();
                errdefer tx.rollback() catch {};
                
                try migration.down(tx);
                try self.removeMigrationRecord(tx, migration.version);
                try tx.commit();
                
                std.log.info("Rollback completed: {s}", .{migration.name});
                return;
            }
        }
        
        return error.MigrationNotFound;
    }
    
    fn ensureMigrationTable(self: *Migrator) !void {
        const sql = 
          \\CREATE TABLE IF NOT EXISTS schema_migrations (
          \\  id BIGSERIAL PRIMARY KEY,
          \\  version VARCHAR(14) UNIQUE NOT NULL,
          \\  name VARCHAR(255) NOT NULL,
          \\  applied_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
          \\);
        try self.db.newRaw(sql, .{}).exec();
    }
};
```
