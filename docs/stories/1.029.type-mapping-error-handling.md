// 基础类型映射
const User = struct {

### 错误处理模式

[Source: docs/functional_spec.md - 错误处理]

```zig
// 1. 错误传播 (最简单)
const user = try findUser(db, id);

// 2. 错误捕获和处理
const user = findUser(db, id) catch |err| switch (err) {
    error.NoRows => {
        std.log.warn("User not found: {d}", .{id});
        return null;
    },
    error.ConnectionFailed => {
        std.log.err("Database connection failed", .{});
        // 重试逻辑
        std.time.sleep(1_000_000_000);  // 1秒
        return try findUser(db, id);
    },
    error.UniqueViolation => {
        std.log.err("Duplicate key violation", .{});
        return error.DuplicateUser;
    },
    else => return err,
};

// 3. errdefer 清理资源
const tx = try db.beginTx();
errdefer tx.rollback() catch {};

try tx.exec("INSERT INTO users ...", .{});
try tx.exec("INSERT INTO profiles ...", .{});

try tx.commit();
```

### ZORM 错误类型

```zig
// 连接错误
error.ConnectionFailed      // 连接失败
error.ConnectionTimeout     // 连接超时
error.ConnectionClosed      // 连接已关闭

// 查询错误
error.QueryFailed           // 查询失败
error.InvalidSQL            // SQL 语法错误
error.PrepareStatementFailed  // 预编译失败

// 数据错误
error.NoRows                // 无结果
error.TooManyRows           // 结果过多
error.ScanError             // 扫描错误
error.TypeMismatch          // 类型不匹配

// 约束错误
error.UniqueViolation       // 唯一约束违反
error.ForeignKeyViolation   // 外键约束违反
error.NotNullViolation      // 非空约束违反
error.CheckViolation        // 检查约束违反

// 事务错误
error.TransactionFailed     // 事务失败
error.DeadlockDetected      // 死锁检测
```

### 错误恢复策略

```zig
// 1. 重试机制 (指数退避)
fn executeWithRetry(
    db: *zorm.DB,
    sql: []const u8,
    max_retries: usize,
) !void {
    var retries: usize = 0;
    while (retries < max_retries) : (retries += 1) {
        db.exec(sql, .{}) catch |err| switch (err) {
            error.ConnectionFailed,
            error.DeadlockDetected => {
                const delay_ms = @as(u64, 100) * (@as(u64, 1) << @intCast(retries));
                std.time.sleep(delay_ms * 1_000_000);
                continue;
            },
            else => return err,
        };
        return;
    }
    return error.MaxRetriesExceeded;
}

// 2. 降级处理
fn getUserData(db: *zorm.DB, id: i32) !?User {
    return db.findUser(id) catch |err| switch (err) {
        error.ConnectionFailed => {
            // 降级到缓存
            return cache.get(id);
        },
        error.NoRows => {
            return null;
        },
        else => return err,
    };
}

// 3. 断路器模式
const CircuitBreaker = struct {
    failure_threshold: usize = 5,
    timeout_ms: u64 = 60_000,
    // ... 实现
};
```

test "类型映射 - 基础类型" {
    const allocator = testing.allocator;
    const db = try zorm.DB.initMock(allocator, .postgresql);
    defer db.deinit();

    const user = User{
        .id = 1,
        .name = "Alice",
        .age = 25,
        .balance = 100.50,
        .is_active = true,
        .created_at = 1609459200,

    // 插入
    var insert = try db.newInsert(User);

    // 查询

    try query.where("id = ?", .{1}).scanOne(&result);

    try testing.expectEqual(user.id, result.id);
    try testing.expectEqualStrings(user.name, result.name);
}

test "NULL 值处理" {
    const allocator = testing.allocator;
    const db = try zorm.DB.initMock(allocator, .postgresql);
    defer db.deinit();

    const UserWithOptional = struct {
        id: i32,
        name: []const u8,
        email: ?[]const u8,
    };

    const user = UserWithOptional{
        .id = 1,
        .name = "Alice",
        .email = null,  // NULL 值
    };

    var insert = try db.newInsert(UserWithOptional);
    try insert.value(user).exec();

    var query = try db.newSelect(UserWithOptional);
    var result: UserWithOptional = undefined;
    try query.where("id = ?", .{1}).scanOne(&result);

    try testing.expect(result.email == null);

test "错误处理 - NoRows" {
    const allocator = testing.allocator;
    const db = try zorm.DB.initMock(allocator, .postgresql);
    defer db.deinit();

    var query = try db.newSelect(User);
    var result: User = undefined;

    const err = query.where("id = ?", .{9999}).scanOne(&result);
    try testing.expectError(error.NoRows, err);

test "错误恢复 - 重试机制" {
    const allocator = testing.allocator;
    const db = try zorm.DB.initMock(allocator, .postgresql);
    defer db.deinit();

    var retries: usize = 0;
    const max_retries: usize = 3;

    while (retries < max_retries) : (retries += 1) {
        db.exec("SELECT * FROM users", .{}) catch |err| switch (err) {
            error.ConnectionFailed => {
                std.time.sleep(100_000_000);  // 100ms
                continue;
            },
            else => return err,
        };
        break;
    }
}
