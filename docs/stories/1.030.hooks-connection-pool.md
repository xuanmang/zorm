### 钩子系统 API

[Source: docs/functional_spec.md - 钩子系统]

#### 钩子接口定义

```zig
/// 查询上下文
pub const QueryContext = struct {
    sql: []const u8,
    params: []const anytype,
    start_time: i64,
    duration_ms: ?u64 = null,
    user_id: ?i32 = null,
    metadata: ?std.StringHashMap([]const u8) = null,
};

/// 查询钩子接口
pub const QueryHook = struct {
    /// 查询前钩子 (可选)
    beforeQuery: ?*const fn(*QueryContext) anyerror!void = null,

    /// 查询后钩子 (可选)
    afterQuery: ?*const fn(*QueryContext, *Result) anyerror!void = null,

    /// 错误钩子 (可选)
    onError: ?*const fn(*QueryContext, anyerror) void = null,
};
```

#### 日志记录钩子

```zig
fn logBefore(ctx: *QueryContext) !void {
    std.log.info("SQL: {s}", .{ctx.sql});
    if (ctx.params.len > 0) {
        std.log.info("Params: {any}", .{ctx.params});
    }
}

fn logAfter(ctx: *QueryContext, result: *Result) !void {
    std.log.info(
        "Rows: {d}, Time: {d}ms",
        .{result.rows_affected, ctx.duration_ms orelse 0}
    );
}

const LoggingHook = QueryHook{
    .beforeQuery = logBefore,
    .afterQuery = logAfter,
};

// 注册钩子
const db_with_logging = try db.withQueryHook(&LoggingHook);
```

#### 性能监控钩子

```zig
const PerformanceHook = struct {
    threshold_ms: u64,

    pub fn beforeQuery(ctx: *QueryContext) !void {
        ctx.start_time = std.time.milliTimestamp();
    }

    pub fn afterQuery(ctx: *QueryContext, result: *Result) !void {
        const now = std.time.milliTimestamp();
        ctx.duration_ms = @intCast(now - ctx.start_time);

        if (ctx.duration_ms.? > threshold_ms) {
            std.log.warn(
                "Slow query detected: {s} ({d}ms)",
                .{ctx.sql, ctx.duration_ms.?}
            );
        }
    }
};

const perf_hook = PerformanceHook{ .threshold_ms = 100 };
const db_with_perf = try db.withQueryHook(&perf_hook.hook());
```

#### 钩子组合

```zig
// 多个钩子链式调用
const db_with_hooks = try db
    .withQueryHook(&LoggingHook)
    .withQueryHook(&PerformanceHook)
    .withQueryHook(&AuditHook);

// 所有钩子按注册顺序触发
try db_with_hooks.exec("SELECT * FROM users", .{});
```

### 连接池 API

[Source: docs/functional_spec.md - 连接池管理]

#### 连接池配置

```zig
pub const PoolConfig = struct {
    /// 最大打开连接数 (0 = 无限制)
    max_open_conns: usize = 0,

    /// 最大空闲连接数
    max_idle_conns: usize = 2,

    /// 连接最大生命周期 (秒, 0 = 永久)
    conn_max_lifetime: u64 = 0,

    /// 连接最大空闲时间 (秒, 0 = 永久)
    conn_max_idle_time: u64 = 0,

    /// 连接等待超时 (毫秒)
    acquire_timeout_ms: u64 = 30_000,
};
```

#### 连接池使用

```zig
// 创建连接池
const pool_config = PoolConfig{
    .max_open_conns = 25,
    .max_idle_conns = 10,
    .conn_max_lifetime = 300,  // 5分钟
    .conn_max_idle_time = 60,  // 1分钟
};

const pool = try Pool.init(allocator, dsn, pool_config);
defer pool.deinit();

// 获取连接
const conn = try pool.acquire();
defer pool.release(conn) catch {};

// 使用连接
const db = try zorm.DB.init(allocator, conn, .postgresql, .{});
defer db.deinit();

try db.exec("SELECT * FROM users", .{});
```

#### 连接池统计

```zig
pub const PoolStats = struct {
    max_open_conns: usize,
    open_conns: usize,      // 当前打开的连接数
    in_use: usize,          // 当前使用中的连接数
    idle: usize,            // 当前空闲的连接数
    wait_count: usize,      // 等待连接的次数
    wait_duration_ms: u64,  // 总等待时长
};

// 获取统计信息
const stats = pool.stats();
std.log.info("Pool Stats:", .{});
std.log.info("  Open: {d}/{d}", .{stats.open_conns, stats.max_open_conns});
std.log.info("  In Use: {d}", .{stats.in_use});
std.log.info("  Idle: {d}", .{stats.idle});
```

#### 连接池最佳实践

```zig
// 1. 根据应用负载调整配置
const pool_config = PoolConfig{
    // 最大连接数 = CPU核心数 * 2 + 磁盘数
    .max_open_conns = @import("builtin").cpu.arch.count() * 2 + 4,

    // 空闲连接数 = 最大连接数的 40%
    .max_idle_conns = 10,

    // 连接生命周期 = 5分钟 (避免连接泄漏)
    .conn_max_lifetime = 300,

    // 空闲超时 = 1分钟 (快速回收)
    .conn_max_idle_time = 60,
};

// 2. 使用 defer 确保连接释放
{
    const conn = try pool.acquire();
    defer pool.release(conn) catch {};

    // 使用连接...
}  // 自动释放

// 3. 长事务使用独立连接
const dedicated_conn = try pool.acquireDedicated();
defer pool.releaseDedicated(dedicated_conn) catch {};
```

## Testing

### 测试策略

```zig
test "钩子系统 - 日志记录" {
    const allocator = testing.allocator;
    const db = try zorm.DB.initMock(allocator, .postgresql);
    defer db.deinit();

    var log_count: usize = 0;
    const TestHook = struct {
        fn before(ctx: *QueryContext) !void {
            log_count += 1;
        }
    };

    const hook = QueryHook{ .beforeQuery = TestHook.before };
    const db_with_hook = try db.withQueryHook(&hook);

    try db_with_hook.exec("SELECT 1", .{});
    try testing.expectEqual(@as(usize, 1), log_count);
}

test "钩子组合 - 链式调用" {
    const allocator = testing.allocator;
    const db = try zorm.DB.initMock(allocator, .postgresql);
    defer db.deinit();

    var hook1_called = false;
    var hook2_called = false;

    const Hook1 = QueryHook{
        .beforeQuery = struct {
            fn call(ctx: *QueryContext) !void {
                hook1_called = true;
            }
        }.call,
    };

    const Hook2 = QueryHook{
        .beforeQuery = struct {
            fn call(ctx: *QueryContext) !void {
                hook2_called = true;
            }
        }.call,
    };

    const db_with_hooks = try db
        .withQueryHook(&Hook1)
        .withQueryHook(&Hook2);

    try db_with_hooks.exec("SELECT 1", .{});

    try testing.expect(hook1_called);
    try testing.expect(hook2_called);
}

test "连接池 - 并发获取" {
    const allocator = testing.allocator;

    const pool_config = PoolConfig{
        .max_open_conns = 5,
        .max_idle_conns = 2,
    };

    const pool = try Pool.init(allocator, "mock_dsn", pool_config);
    defer pool.deinit();

    // 并发获取连接
    var conns: [5]*Conn = undefined;
    for (&conns) |*conn| {
        conn.* = try pool.acquire();
    }

    // 验证池统计
    const stats = pool.stats();
    try testing.expectEqual(@as(usize, 5), stats.in_use);
    try testing.expectEqual(@as(usize, 0), stats.idle);

    // 释放连接
    for (conns) |conn| {
        try pool.release(conn);
    }
}

test "连接池 - 生命周期管理" {
    const allocator = testing.allocator;

    const pool_config = PoolConfig{
        .max_open_conns = 2,
        .conn_max_lifetime = 1,  // 1秒
    };

    const pool = try Pool.init(allocator, "mock_dsn", pool_config);
    defer pool.deinit();

    const conn = try pool.acquire();

    // 等待超过生命周期
    std.time.sleep(1_500_000_000);  // 1.5秒

    try pool.release(conn);

    // 连接应该被关闭并重新创建
    const new_conn = try pool.acquire();
    try testing.expect(conn != new_conn);
}
```
