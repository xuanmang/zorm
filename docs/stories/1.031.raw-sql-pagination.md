#### 基础原始 SQL

```zig
// 简单查询
try db.newRaw("SELECT * FROM users", .{}).exec();
```

#### 复杂 SQL 查询

```zig
// 复杂聚合
const sql =
  \\SELECT
  \\  u.id,
  \\  u.name,
  \\  COUNT(p.id) as post_count,
  \\  AVG(LENGTH(p.content)) as avg_content_length
  \\FROM users u
  \\LEFT JOIN posts p ON p.user_id = u.id
  \\GROUP BY u.id, u.name
  \\HAVING COUNT(p.id) > ?
  \\ORDER BY post_count DESC;

try db.newRaw(sql, .{10}).exec();

// CTE (WITH 子句)
const sql =
  \\WITH active_users AS (
  \\  SELECT * FROM users WHERE last_login > NOW() - INTERVAL '7 days'
  \\),
  \\top_posters AS (
  \\  SELECT user_id, COUNT(*) as post_count
  \\  FROM posts
  \\  GROUP BY user_id
  \\  ORDER BY post_count DESC
  \\  LIMIT 10
  \\)
  \\SELECT u.*, tp.post_count
  \\FROM active_users u
  \\JOIN top_posters tp ON tp.user_id = u.id;

try db.newRaw(sql, .{}).exec();
```

### 分页查询模式

[Source: docs/functional_spec.md - 分页查询]

#### OFFSET/LIMIT 分页

```zig
// 基础分页
const page: usize = 2;
const page_size: usize = 20;
const offset = (page - 1) * page_size;

var query = try db.newSelect(User);
try query
    .orderBy("id", .asc)
    .limit(page_size)
    .offset(offset);

// 获取总数 (需要额外查询)
var count_query = try db.newSelect(User);
try count_query.column("COUNT(*) as total");
const total = try count_query.scanOne();

// 计算分页元数据
const total_pages = (total + page_size - 1) / page_size;
const has_next = page < total_pages;
const has_prev = page > 1;
```

#### Cursor-based 分页 (推荐)

```zig
// 第一页
var query = try db.newSelect(User);
try query
    .orderBy("id", .asc)
    .limit(page_size);

var users: std.ArrayList(User) = .{};
defer users.deinit(allocator);
try query.scan(allocator, &users);

// 获取最后一条的 cursor (ID)
const last_cursor = if (users.items.len > 0)
    users.items[users.items.len - 1].id
else
    null;

// 下一页 (使用 cursor)
if (last_cursor) |cursor| {
    var next_query = try db.newSelect(User);
    try next_query
        .where("id > ?", .{cursor})
        .orderBy("id", .asc)
        .limit(page_size);
}

// 上一页 (反向查询)
if (first_cursor) |cursor| {
    var prev_query = try db.newSelect(User);
    try prev_query
        .where("id < ?", .{cursor})
        .orderBy("id", .desc)  // 反向排序
        .limit(page_size);

    // 结果需要反转顺序
    std.mem.reverse(User, users.items);
}
```

#### 分页辅助函数

```zig
pub const PaginationResult = struct {
    items: []const User,
    total: usize,
    page: usize,
    page_size: usize,
    total_pages: usize,
    has_next: bool,
    has_prev: bool,

    pub fn deinit(self: *PaginationResult, allocator: Allocator) void {
        allocator.free(self.items);
    }
};

pub fn paginate(
    allocator: Allocator,
    db: *zorm.DB,
    comptime T: type,
    page: usize,
    page_size: usize,
) !PaginationResult {
    // 获取总数
    var count_query = try db.newSelect(T);
    try count_query.column("COUNT(*) as total");
    const total = try count_query.scanOne();

    // 获取数据
    const offset = (page - 1) * page_size;
    var query = try db.newSelect(T);
    try query.limit(page_size).offset(offset);

    var items: std.ArrayList(T) = .{};
    defer items.deinit(allocator);
    try query.scan(allocator, &items);

    // 计算元数据
    const total_pages = (total + page_size - 1) / page_size;

    return PaginationResult{
        .items = try items.toOwnedSlice(),
        .total = total,
        .page = page,
        .page_size = page_size,
        .total_pages = total_pages,
        .has_next = page < total_pages,
        .has_prev = page > 1,
    };
}
```

#### Cursor-based 分页辅助函数

```zig
pub const CursorPaginationResult = struct {
    items: []const User,
    next_cursor: ?i32,
    prev_cursor: ?i32,
    has_next: bool,
    has_prev: bool,

    pub fn deinit(self: *CursorPaginationResult, allocator: Allocator) void {
        allocator.free(self.items);
    }
};

pub fn paginateByCursor(
    allocator: Allocator,
    db: *zorm.DB,
    comptime T: type,
    after: ?i32,  // cursor
    limit: usize,
) !CursorPaginationResult {
    var query = try db.newSelect(T);

    if (after) |cursor| {
        try query.where("id > ?", .{cursor});
    }

    try query.orderBy("id", .asc).limit(limit + 1);  // +1 检测是否有下一页

    var items: std.ArrayList(T) = .{};
    defer items.deinit(allocator);
    try query.scan(allocator, &items);

    const has_next = items.items.len > limit;
    if (has_next) {
        _ = items.pop();  // 移除多余的一条
    }

    const next_cursor = if (has_next and items.items.len > 0)
        items.items[items.items.len - 1].id
    else
        null;

    const prev_cursor = if (items.items.len > 0)
        items.items[0].id
    else
        null;

    return CursorPaginationResult{
        .items = try items.toOwnedSlice(),
        .next_cursor = next_cursor,
        .prev_cursor = prev_cursor,
        .has_next = has_next,
        .has_prev = after != null,
    };
}
```

### 性能优化技巧

#### OFFSET/LIMIT 性能问题

```text
数据量      OFFSET/LIMIT 耗时    Cursor-based 耗时
------------------------------------------------------
1万条       10ms                5ms
10万条      100ms               6ms
100万条     1000ms              7ms
1000万条    10000ms             8ms
```

#### 优化建议

1. **使用 Cursor-based 分页**: 性能稳定,不受数据量影响
2. **避免深分页**: OFFSET 越大,性能越差
3. **添加索引**: 在排序字段上创建索引
4. **缓存总数**: 总数查询代价高,考虑缓存
5. **限制最大页数**: 防止用户跳转到极远的页面

```zig
// ✅ 好的做法: Cursor-based
var query = try db.newSelect(User);
try query.where("id > ?", .{last_id}).limit(20);

// ❌ 差的做法: 深 OFFSET
var query = try db.newSelect(User);
try query.offset(999_980).limit(20);  // 跳到第 50000 页
```
