# Story 1.2: Type-Safe SELECT Query Builder (Basic)

**Epic**: 1 - Foundation & Query Builder Core
**Story**: 1.2
**Status**: Ready for Review

---

## Story

**As a** Zig developer,  
**I want** 使用类型安全的 SELECT 查询构建器检索数据,  
**so that** 我能够以编译时验证的方式查询 PostgreSQL 数据库，避免运行时错误。

---

## Acceptance Criteria

1. **AC1.2.1**: 提供 `db.newSelect(T)` API 创建 SELECT 查询构建器，T 为目标 Zig 结构体类型
2. **AC1.2.2**: 支持 `.where(condition, args)` 方法添加 WHERE 条件，参数使用元组自动绑定（防止 SQL 注入）
3. **AC1.2.3**: 支持 `.orderBy(column, direction)` 方法指定排序，direction 为枚举 `.asc` 或 `.desc`
4. **AC1.2.4**: 支持 `.limit(n)` 和 `.offset(n)` 方法实现分页
5. **AC1.2.5**: 提供 `.scan(dest)` 方法执行查询并将结果扫描到 `std.ArrayList(T)` 中
6. **AC1.2.6**: 提供 `.scanOne()` 方法查询单行，返回 `!T` 类型（无结果返回 error.NoRows）
7. **AC1.2.7**: 查询构建器支持链式调用（fluent interface），提升代码可读性
8. **AC1.2.8**: 使用示例验证完整工作流程

### SelectQuery 架构设计
[Source: docs/architecture/architecture-core.md#2-query-builders-查询构建器]

**完整 API 设计**:
```zig
pub fn SelectQuery(comptime T: type) type {
    return struct {
        const Self = @This();

        allocator: Allocator,
        db: *DB,
        columns: std.ArrayList([]const u8),
        table_name: []const u8,
        where_clauses: std.ArrayList(WhereClause),
        order_by_clauses: std.ArrayList(OrderByClause),
        limit_value: ?usize,
        offset_value: ?usize,

        /// 添加 WHERE 条件 (AND)
        pub fn where(self: *Self, condition: []const u8, args: anytype) !*Self;

        /// 指定排序
        pub fn orderBy(self: *Self, column: []const u8, direction: OrderDirection) *Self;

        /// 限制结果数量
        pub fn limit(self: *Self, n: usize) *Self;

        /// 偏移量
        pub fn offset(self: *Self, n: usize) *Self;

        /// 构建 SQL (不执行)
        pub fn build(self: *Self) ![]const u8;

        /// 执行查询并扫描到 ArrayList
        pub fn scan(self: *Self, dest: *std.ArrayList(T)) !void;

        /// 执行查询并返回单行
        pub fn scanOne(self: *Self) !T;

        /// 释放资源
        pub fn deinit(self: *Self) void;
    };
}
```

**关键设计要点**:
1. **泛型类型 T**: 在编译时确定目标结构体，实现类型安全
2. **链式调用**: 所有修改方法返回 `*Self`，支持流畅 API
3. **分离构建和执行**: `build()` 生成 SQL，`scan()` 执行查询
4. **参数绑定**: 使用 PostgreSQL `$1, $2, ...` 占位符防止 SQL 注入

### 类型系统实现
[Source: docs/architecture/architecture-core.md#3-type-system-类型系统]

**核心类型映射函数**:
```zig
/// Zig 类型到 SQL 类型映射
pub fn zigToSQLType(comptime T: type) []const u8 {
    return switch (@typeInfo(T)) {
        .Int => |info| {
            if (info.bits == 64) return "BIGINT";
            if (info.bits == 32) return "INTEGER";
            return "INTEGER";
        },
        .Bool => "BOOLEAN",
        .Float => |info| {
            if (info.bits == 64) return "DOUBLE PRECISION";
            return "REAL";
        },
        .Pointer => |info| {
            if (info.child == u8) return "TEXT";
            @compileError("不支持的指针类型: " ++ @typeName(T));
        },
        .Optional => |info| {
            return zigToSQLType(info.child); // 可选类型递归映射
        },
        else => @compileError("不支持的 Zig 类型: " ++ @typeName(T)),
    };
}

/// 获取表名
pub fn getTableName(comptime T: type) []const u8 {
    if (@hasDecl(T, "table_name")) {
        return T.table_name;
    }
    // 否则使用类型名的小写形式
    return toLowerSnakeCase(@typeName(T));
}

/// 反射结构体字段
pub fn getFields(comptime T: type) []const std.builtin.Type.StructField {
    const type_info = @typeInfo(T);
    if (type_info != .Struct) {
        @compileError(@typeName(T) ++ " 不是结构体类型");
    }
    return type_info.Struct.fields;
}
```

**使用示例**:
```zig
const User = struct {
    id: i64,              // 映射为 BIGINT
    name: []const u8,     // 映射为 TEXT
    email: ?[]const u8,   // 映射为 TEXT (NULL 允许)
    age: u32,             // 映射为 INTEGER
    is_active: bool,      // 映射为 BOOLEAN

    pub const table_name = "users"; // 自定义表名
};

// 编译时推断
comptime {
    assert(getTableName(User) == "users");
    assert(zigToSQLType(i64) == "BIGINT");
}
```

### 内存管理模式
[Source: docs/architecture/coding-standards.md#内存管理规范]

**SelectQuery 内存管理**:
```zig
pub fn newSelect(self: *DB, comptime T: type) !SelectQuery(T) {
    var query = SelectQuery(T){
        .allocator = self.allocator,
        .db = self,
        .columns = .{},
        .table_name = getTableName(T),
        .where_clauses = .{},
        .order_by_clauses = .{},
        .limit_value = null,
        .offset_value = null,
    };
    
    // 预分配容量优化
    try query.where_clauses.ensureTotalCapacity(self.allocator, 4);
    try query.order_by_clauses.ensureTotalCapacity(self.allocator, 2);
    
    return query;
}

pub fn deinit(self: *Self) void {
    self.columns.deinit(self.allocator);
    self.where_clauses.deinit(self.allocator);
    self.order_by_clauses.deinit(self.allocator);
}
```

**Arena Allocator 优化**:
```zig
pub fn build(self: *Self) ![]const u8 {
    var arena = std.heap.ArenaAllocator.init(self.allocator);
    defer arena.deinit();

    const arena_allocator = arena.allocator();

    var buf: std.ArrayList(u8) = .{};
    defer buf.deinit(arena_allocator);

    // 构建 SQL...
    try buf.appendSlice(arena_allocator, "SELECT * FROM ");
    try buf.appendSlice(arena_allocator, self.table_name);

    // 最终结果使用主 allocator
    return self.allocator.dupe(u8, buf.items);
}
```

```zig
pub const ZormError = error{

    ConnectionClosed,    // 数据库连接已关闭

```

**错误传播模式**:
```zig
pub fn scanOne(self: *Self) !T {
    var temp: std.ArrayList(T) = .{};
    defer temp.deinit(self.allocator);

    try self.scan(&temp); // 传播错误

    if (temp.items.len == 0) {
        return error.NoRows;
    }
    if (temp.items.len > 1) {
        return error.MultipleRows;
    }

    return temp.items[0];
}
```

```zig
    // 连接真实数据库
```
