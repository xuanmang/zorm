### 列选择实现
[Source: docs/prd.md - Story 1.3 AC]

**API 设计**:
```zig
pub fn SelectQuery(comptime T: type) type {
    return struct {
        // ... 现有字段
        columns: std.ArrayList([]const u8),  // 已在 Story 1.2 中定义
        distinct_value: bool,                 // 新增字段

        /// 选择特定列
        pub fn column(self: *Self, name: []const u8) !*Self {
            try self.columns.append(self.allocator, name);
            return self;
        }

        /// 启用 DISTINCT
        pub fn setDistinct(self: *Self) *Self {
            self.distinct_value = true;
            return self;
        }

        /// 返回查询结果数量
        pub fn count(self: *Self) !usize {
            // 构建 SELECT COUNT(*) 查询
            var count_query = try self.buildCountQuery();
            defer self.allocator.free(count_query);

            // 执行查询
            const result = try self.db.conn.execute(count_query, self.args);
            defer result.deinit();

            // 提取 count 值
            return try result.getInt(usize, 0, 0);
        }
    };
}
```

### SQL 生成逻辑

**列选择 SQL 生成**:
```zig
pub fn build(self: *Self) ![]const u8 {
    var arena = std.heap.ArenaAllocator.init(self.allocator);
    defer arena.deinit();
    const arena_allocator = arena.allocator();

    var buf: std.ArrayList(u8) = .{};
    defer buf.deinit(arena_allocator);

    // SELECT 子句
    try buf.appendSlice(arena_allocator, "SELECT ");
    
    // DISTINCT
    if (self.distinct_value) {
        try buf.appendSlice(arena_allocator, "DISTINCT ");
    }

    // 列选择
    if (self.columns.items.len == 0) {
        try buf.appendSlice(arena_allocator, "*");
    } else {
        for (self.columns.items, 0..) |col, i| {
            if (i > 0) {
                try buf.appendSlice(arena_allocator, ", ");
            }
            try buf.appendSlice(arena_allocator, col);
        }
    }

    // FROM 子句
    try buf.appendSlice(arena_allocator, " FROM ");
    try buf.appendSlice(arena_allocator, self.table_name);

    // WHERE, ORDER BY, LIMIT, OFFSET（Story 1.2 已实现）
    // ...

    return self.allocator.dupe(u8, buf.items);
}
```

**SQL 示例**:
```sql
-- 默认（无列选择）
SELECT * FROM users WHERE age > $1

-- 列选择
SELECT id, name, email FROM users WHERE age > $1

-- DISTINCT
SELECT DISTINCT email FROM users

-- 列选择 + DISTINCT
SELECT DISTINCT status FROM users WHERE is_active = $1

-- COUNT
SELECT COUNT(*) FROM users WHERE age > $1
```

### Count 实现

**Count 查询构建**:
```zig
fn buildCountQuery(self: *Self) ![]const u8 {
    var arena = std.heap.ArenaAllocator.init(self.allocator);
    defer arena.deinit();
    const arena_allocator = arena.allocator();

    var buf: std.ArrayList(u8) = .{};
    defer buf.deinit(arena_allocator);

    // SELECT COUNT(*)
    try buf.appendSlice(arena_allocator, "SELECT COUNT(");
    
    if (self.distinct_value) {
        try buf.appendSlice(arena_allocator, "DISTINCT ");
    }
    
    if (self.columns.items.len == 0) {
        try buf.appendSlice(arena_allocator, "*");
    } else {
        // COUNT(DISTINCT col1, col2) 在 PostgreSQL 中不支持多列
        // 使用 COUNT(DISTINCT (col1, col2))
        if (self.columns.items.len > 1 and self.distinct_value) {
            try buf.appendSlice(arena_allocator, "(");
            for (self.columns.items, 0..) |col, i| {
                if (i > 0) try buf.appendSlice(arena_allocator, ", ");
                try buf.appendSlice(arena_allocator, col);
            }
            try buf.appendSlice(arena_allocator, ")");
        } else {
            try buf.appendSlice(arena_allocator, self.columns.items[0]);
        }
    }
    
    try buf.appendSlice(arena_allocator, ") FROM ");
    try buf.appendSlice(arena_allocator, self.table_name);

    // 添加 WHERE 条件（如果有）
    // ...

    return self.allocator.dupe(u8, buf.items);
}
```

### 结构体字段映射

**部分列选择的结果映射**:

当用户只选择部分列时，需要定义对应的结构体：

```zig
const User = struct {
    id: i64,
    name: []const u8,
    email: []const u8,
    age: u32,
};

// 只选择 id 和 name
const UserIdName = struct {
    id: i64,
    name: []const u8,
};

var users: std.ArrayList(UserIdName) = .{};
defer users.deinit(allocator);

var query = try db.newSelect(UserIdName);
defer query.deinit();

try query
    .column("id")
    .column("name")
    .where("age > ?", .{18})
    .scan(&users);
```

**注意**:
- 列选择必须与目标结构体字段匹配
- 字段顺序不重要（通过字段名匹配）
- 如果选择的列与结构体不匹配，会产生运行时错误

### 性能优化

**列选择的性能优势**:
- 减少网络传输（只传输需要的列）
- 减少内存占用（只存储需要的字段）
- 提升查询性能（PostgreSQL 只需读取选择的列）

**DISTINCT 性能考虑**:
- DISTINCT 需要排序或哈希操作，性能开销较大
- 仅在必要时使用 DISTINCT
- 如果可能，通过 GROUP BY 替代 DISTINCT

**COUNT 性能优化**:
- `COUNT(*)` 通常比 `COUNT(column)` 快
- PostgreSQL 对 `COUNT(*)` 有优化
- 对于大表，考虑使用近似计数或缓存

test "SelectQuery: column selection" {
