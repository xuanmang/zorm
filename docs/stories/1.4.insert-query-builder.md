# Story 1.4: Type-Safe INSERT Query Builder (Single Row)

**Epic**: 1 - Foundation & Query Builder Core
**Story**: 1.4
**Status**: Ready for Review

---

## Story

**As a** Zig developer,  
**I want** 使用类型安全的 INSERT 构建器插入单行数据,  
**so that** 我能够以类型安全的方式向数据库添加数据，并利用 PostgreSQL RETURNING 获取插入结果。

### InsertQuery 结构设计

[Source: docs/architecture/architecture-core.md#InsertQuery]

**核心结构**:
```zig
pub fn InsertQuery(comptime T: type) type {
    return struct {
        const Self = @This();

        allocator: Allocator,
        db: *DB,
        table_name: []const u8,
        columns: std.ArrayList([]const u8),
        values_list: std.ArrayList([]const QueryArg),
        returning_columns: ?[]const []const u8,

        /// 插入单行数据
        pub fn value(self: *Self, item: T) !*Self {
            // 使用 comptime 反射提取字段值
            const fields = @typeInfo(T).Struct.fields;
            var args = std.ArrayList(QueryArg).init(self.allocator);
            
            inline for (fields) |field| {
                const field_value = @field(item, field.name);
                const arg = try valueToQueryArg(field_value, field.type);
                try args.append(self.allocator, arg);
            }
            
            try self.values_list.append(self.allocator, try args.toOwnedSlice());
            return self;
        }

        /// 构建 INSERT SQL
        pub fn build(self: *Self) ![]const u8 {
            var arena = std.heap.ArenaAllocator.init(self.allocator);
            defer arena.deinit();
            const arena_allocator = arena.allocator();

            var buf: std.ArrayList(u8) = .{};
            defer buf.deinit(arena_allocator);

            // INSERT INTO table_name
            try buf.appendSlice(arena_allocator, "INSERT INTO ");
            try buf.appendSlice(arena_allocator, self.table_name);

            // (col1, col2, col3)
            try buf.appendSlice(arena_allocator, " (");
            for (self.columns.items, 0..) |col, i| {
                if (i > 0) try buf.appendSlice(arena_allocator, ", ");
                try buf.appendSlice(arena_allocator, col);
            }
            try buf.appendSlice(arena_allocator, ")");

            // VALUES ($1, $2, $3)
            try buf.appendSlice(arena_allocator, " VALUES (");
            for (0..self.columns.items.len) |i| {
                if (i > 0) try buf.appendSlice(arena_allocator, ", ");
                const placeholder = self.db.dialect.placeholder(i + 1);
                try buf.appendSlice(arena_allocator, placeholder);
            }
            try buf.appendSlice(arena_allocator, ")");

            // RETURNING col1, col2
            if (self.returning_columns) |cols| {
                try buf.appendSlice(arena_allocator, " RETURNING ");
                if (cols.len == 1 and std.mem.eql(u8, cols[0], "*")) {
                    try buf.appendSlice(arena_allocator, "*");
                } else {
                    for (cols, 0..) |col, i| {
                        if (i > 0) try buf.appendSlice(arena_allocator, ", ");
                        try buf.appendSlice(arena_allocator, col);
                    }
                }
            }

            return self.allocator.dupe(u8, buf.items);
        }

        /// 执行插入
        pub fn exec(self: *Self) !InsertResult {
            const sql = try self.build();
            defer self.allocator.free(sql);

            // 钩子：beforeQuery
            if (self.db.hooks) |hook| {
                if (hook.beforeQuery) |beforeFn| {
                    try beforeFn(sql, self.values_list.items[0]);
                }
            }

            const start_time = std.time.nanoTimestamp();
            const result = try self.db.conn.execute(sql, self.values_list.items[0]);
            const duration = std.time.nanoTimestamp() - start_time;

            // 钩子：afterQuery
            if (self.db.hooks) |hook| {
                if (hook.afterQuery) |afterFn| {
                    try afterFn(sql, duration, null);
                }
            }

            return InsertResult{
                .rows_affected = result.rows_affected,
                .last_insert_id = null, // PostgreSQL 需要通过 RETURNING 获取
            };
        }

        /// 执行插入并返回插入的数据
        pub fn execReturning(self: *Self, dest: *std.ArrayList(T)) !void {
            const sql = try self.build();
            defer self.allocator.free(sql);

            const result = try self.db.conn.execute(sql, self.values_list.items[0]);
            defer result.deinit();

            // 复用 SelectQuery 的结果扫描逻辑
            try scanResult(T, result, dest, self.allocator);
        }

        pub fn deinit(self: *Self) void {
            self.columns.deinit(self.allocator);
            for (self.values_list.items) |args| {
                self.allocator.free(args);
            }
            self.values_list.deinit(self.allocator);
        }
    };
}
```

**值到 QueryArg 转换**:
```zig
fn valueToQueryArg(value: anytype, comptime T: type) !QueryArg {
    return switch (@typeInfo(T)) {
        .Int => QueryArg{ .int = @intCast(value) },
        .Bool => QueryArg{ .bool = value },
        .Pointer => |info| {
            if (info.child == u8) {
                return QueryArg{ .string = value };
            }
            @compileError("不支持的指针类型");
        },
        .Optional => |info| {
            if (value) |v| {
                return valueToQueryArg(v, info.child);
            } else {
                return QueryArg{ .null = {} };
            }
        },
        else => @compileError("不支持的类型: " ++ @typeName(T)),
    };
}
```

### PostgreSQL RETURNING 子句

[Source: docs/prd.md - FR3]

**RETURNING 特性**:
- PostgreSQL 独有特性（MySQL 不支持）
- 返回插入的行数据，避免额外查询
- 常用于获取自增 ID 或数据库生成的默认值
- 语法：`RETURNING *` 或 `RETURNING col1, col2`

**使用场景**:
```zig
// 场景1：获取自增 ID
var users: std.ArrayList(User) = .{};
defer users.deinit(allocator);

try insert
    .value(user)
    .setReturning(&[_][]const u8{"id"})
    .execReturning(&users);

const inserted_id = users.items[0].id;

// 场景2：获取数据库默认值
try insert
    .value(user)
    .setReturning(&[_][]const u8{"*"})
    .execReturning(&users);

const created_at = users.items[0].created_at;
```

### 内存管理

[Source: docs/architecture/coding-standards.md#内存管理规范]

**Allocator 模式**:
- 所有动态内存通过传入的 allocator 分配
- values_list 存储多个参数数组（为 Story 1.5 批量插入准备）
- 使用 Arena Allocator 优化 SQL 构建
- deinit() 必须释放所有分配的内存

**错误处理**:
- 所有可能失败的操作返回 `!T`
- 使用 errdefer 处理错误路径资源清理
- 内存分配失败返回 error.OutOfMemory

### 与 Bun ORM 对齐

[Source: docs/prd.md - Background Context]

**Bun ORM INSERT API**:
```typescript
const user = await db.insert(users).values({
  name: 'John',
  email: 'john@example.com'
}).returning();
```

**ZORM 等价 API**:
```zig
var inserted: std.ArrayList(User) = .{};
defer inserted.deinit(allocator);

try db.newInsert(User)
    .value(user)
    .setReturning(&[_][]const u8{"*"})
    .execReturning(&inserted);
```

**关键差异**:
- Zig 需要显式内存管理（allocator）
- Zig 需要显式错误处理（try）
- Zig 使用 defer 确保资源释放
- API 模式保持一致（链式调用）
