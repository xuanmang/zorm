# Story 1.5: Batch INSERT Support

**Epic**: 1 - Foundation & Query Builder Core
**Story**: 1.5
**Status**: Ready for Review

## Story

**As a** Zig developer,  
**I want** 批量插入多行数据以提升性能,  
**so that** 我能够高效地向数据库添加大量数据。

1. **AC1.5.1**: 提供 `.values(items)` 方法接受 `[]const T` 切片进行批量插入

3. **AC1.5.3**: 批量插入支持 RETURNING 子句，返回所有插入行的数据
4. **AC1.5.4**: 性能测试验证批量插入比循环单行插入快至少 10 倍
5. **AC1.5.5**: 使用示例验证完整工作流程

### 批量插入 SQL 生成

[Source: docs/prd.md - Story 1.5 AC]

**多行 VALUES 语法**:
```sql
-- 单行（Story 1.4）
INSERT INTO users (name, email, age) 
VALUES ($1, $2, $3)

-- 3 行批量插入（Story 1.5）
INSERT INTO users (name, email, age) 
VALUES 
  ($1, $2, $3),
  ($4, $5, $6),
  ($7, $8, $9)

-- 带 RETURNING
INSERT INTO users (name, email, age) 
VALUES ($1, $2, $3), ($4, $5, $6) 
RETURNING *
```

### values() 方法实现

[Source: docs/architecture/architecture-core.md#InsertQuery]

```zig
/// 批量插入多行数据
pub fn values(self: *Self, items: []const T) !*Self {
    // 预分配容量
    try self.values_list.ensureTotalCapacity(
        self.allocator, 
        self.values_list.items.len + items.len
    );

    // 提取列定义（从第一个 item）
    if (self.columns.items.len == 0) {
        const fields = comptime getFields(T);
        inline for (fields) |field| {
            try self.columns.append(self.allocator, field.name);
        }
    }

    // 遍历所有行，提取字段值
    for (items) |item| {
        var args = std.ArrayList(QueryArg).init(self.allocator);
        errdefer args.deinit(self.allocator);

        const fields = comptime getFields(T);
        inline for (fields) |field| {
            const field_value = @field(item, field.name);
            const arg = try valueToQueryArg(field_value, field.type);
            try args.append(self.allocator, arg);
        }

        try self.values_list.append(
            self.allocator, 
            try args.toOwnedSlice()
        );
    }

    return self;
}
```

### 多行 SQL 构建逻辑

```zig
pub fn build(self: *Self) ![]const u8 {
    var arena = std.heap.ArenaAllocator.init(self.allocator);
    defer arena.deinit();
    const arena_allocator = arena.allocator();

    // 预估 SQL 长度并预分配
    const estimated_size = estimateSQLSize(self);
    var buf: std.ArrayList(u8) = .{};
    try buf.ensureTotalCapacity(arena_allocator, estimated_size);
    defer buf.deinit(arena_allocator);

    // INSERT INTO table_name (col1, col2, col3)
    try buf.appendSlice(arena_allocator, "INSERT INTO ");
    try buf.appendSlice(arena_allocator, self.table_name);
    try buf.appendSlice(arena_allocator, " (");
    for (self.columns.items, 0..) |col, i| {
        if (i > 0) try buf.appendSlice(arena_allocator, ", ");
        try buf.appendSlice(arena_allocator, col);
    }
    try buf.appendSlice(arena_allocator, ") VALUES ");

    // 多行 VALUES
    var param_index: usize = 1;
    for (self.values_list.items, 0..) |row_args, row_idx| {
        if (row_idx > 0) {
            try buf.appendSlice(arena_allocator, ", ");
        }
        
        try buf.appendSlice(arena_allocator, "(");
        for (0..row_args.len) |col_idx| {
            if (col_idx > 0) {
                try buf.appendSlice(arena_allocator, ", ");
            }
            const placeholder = self.db.dialect.placeholder(param_index);
            try buf.appendSlice(arena_allocator, placeholder);
            param_index += 1;
        }
        try buf.appendSlice(arena_allocator, ")");
    }

    // RETURNING 子句
    if (self.returning_columns) |cols| {
        try buf.appendSlice(arena_allocator, " RETURNING ");
        if (cols.len == 1 and std.mem.eql(u8, cols[0], "*")) {
            try buf.appendSlice(arena_allocator, "*");
        } else {
            for (cols, 0..) |col, i| {
                if (i > 0) try buf.appendSlice(arena_allocator, ", ");
                try buf.appendSlice(arena_allocator, col);
            }
        }
    }

    return self.allocator.dupe(u8, buf.items);
}
```

### 参数展平逻辑

**批量插入需要将多行参数展平为单个数组**:

```zig
fn flattenArgs(self: *Self) ![]const QueryArg {
    const total_args = self.values_list.items.len * self.columns.items.len;
    var flat_args = try self.allocator.alloc(QueryArg, total_args);
    errdefer self.allocator.free(flat_args);

    var idx: usize = 0;
    for (self.values_list.items) |row_args| {
        for (row_args) |arg| {
            flat_args[idx] = arg;
            idx += 1;
        }
    }

    return flat_args;
}
```

**示例**:
```zig
// 输入：3 行 x 3 列
values_list = [
    [arg1, arg2, arg3],  // 第 1 行
    [arg4, arg5, arg6],  // 第 2 行
    [arg7, arg8, arg9],  // 第 3 行
]

// 输出：展平为 9 个参数
flat_args = [arg1, arg2, arg3, arg4, arg5, arg6, arg7, arg8, arg9]
// 对应占位符：[$1, $2, $3, $4, $5, $6, $7, $8, $9]
```

### 批量大小限制

[Source: docs/architecture/tech-stack.md - PostgreSQL]

**PostgreSQL 限制**:
- 最大参数数量：65535 (`BIND_MAX_ARGS`)
- 计算公式：`max_rows = 65535 / column_count`
- 示例：
  - 3 列：最大 21845 行
  - 10 列：最大 6553 行
  - 50 列：最大 1310 行

**建议策略**:
```zig
pub const MAX_BATCH_SIZE = 1000; // 保守值，适用于大多数场景

pub fn values(self: *Self, items: []const T) !*Self {
    if (items.len > MAX_BATCH_SIZE) {
        return error.BatchSizeTooLarge;
    }

    const column_count = comptime getFields(T).len;
    const total_params = items.len * column_count;
    if (total_params > 65535) {
        return error.ExceedsPostgreSQLParamLimit;
    }

    // ... 执行批量插入
}
```

### 性能优化策略

[Source: docs/prd.md - NFR7, AC1.5.4]

**批量插入性能优势**:
1. **减少网络往返**：1 次 vs N 次
2. **减少 SQL 解析**：1 次 vs N 次
3. **减少事务开销**：1 次提交 vs N 次提交
4. **PostgreSQL 内部优化**：批量写入比单行写入效率高

**内存优化**:
```zig
// 预分配容量
try values_list.ensureTotalCapacity(allocator, expected_rows);

// 预估 SQL 大小
fn estimateSQLSize(self: *Self) usize {
    const base_size = 100; // INSERT INTO table_name ...
    const cols_size = self.columns.items.len * 20; // 列名
    const row_size = self.columns.items.len * 5; // 每个占位符 "$123, "
    const total_rows = self.values_list.items.len;
    return base_size + cols_size + (row_size * total_rows);
}
```

**性能目标**:
- 批量插入 1000 行 vs 循环单行插入 1000 次
- 性能比率：≥ 10x
- 开销：< pg.zig 原生批量插入的 1.05 倍

### 事务集成

**批量插入应在事务中执行**:
```zig
const tx = try db.beginTx();
errdefer tx.rollback() catch {};

var insert = try tx.newInsert(User);
defer insert.deinit();

try insert.values(&users).exec();
try tx.commit();
```

**优势**:
- 原子性：全部成功或全部失败
- 性能：减少 fsync 次数
- 一致性：避免部分插入

### 性能基准测试

[Source: docs/prd.md - AC1.5.4]

```zig
// benchmarks/insert_bench.zig
const std = @import("std");
const zorm = @import("zorm");

pub fn main() !void {
    const allocator = std.heap.page_allocator;

    var pool = try pg.Pool.init(allocator, .{});
    defer pool.deinit();

    var db = try zorm.DB.init(allocator, &pool, .postgresql, .{});
    defer db.deinit();

    const User = struct {
        name: []const u8,
        email: []const u8,
        pub const table_name = "bench_users";
    };

    // 准备 1000 行数据
    var users: [1000]User = undefined;
    for (&users, 0..) |*user, i| {
        user.* = User{
            .name = try std.fmt.allocPrint(allocator, "User{d}", .{i}),
            .email = try std.fmt.allocPrint(allocator, "user{d}@test.com", .{i}),
        };
    }

    // 基准测试 1: 循环单行插入
    var timer = try std.time.Timer.start();
    const start1 = timer.read();
    
    for (users) |user| {
        var insert = try db.newInsert(User);
        defer insert.deinit();
        _ = try insert.value(user).exec();
    }
    
    const end1 = timer.read();
    const duration1 = end1 - start1;

    // 清理
    try db.exec("TRUNCATE bench_users", .{});

    // 基准测试 2: 批量插入
    const start2 = timer.read();
    
    var batch_insert = try db.newInsert(User);
    defer batch_insert.deinit();
    _ = try batch_insert.values(&users).exec();
    
    const end2 = timer.read();
    const duration2 = end2 - start2;

    // 计算性能比率
    const ratio = @as(f64, @floatFromInt(duration1)) / @as(f64, @floatFromInt(duration2));

    std.debug.print("\n=== Batch Insert Benchmark ===\n", .{});
    std.debug.print("Single row loop: {d}ms\n", .{duration1 / 1_000_000});
    std.debug.print("Batch insert:    {d}ms\n", .{duration2 / 1_000_000});
    std.debug.print("Performance ratio: {d:.2}x\n", .{ratio});
    std.debug.print("Target: >= 10x\n", .{});
    
    if (ratio >= 10.0) {
        std.debug.print("✅ PASS: Batch insert is {d:.2}x faster\n", .{ratio});
    } else {
        std.debug.print("❌ FAIL: Batch insert is only {d:.2}x faster\n", .{ratio});
        return error.PerformanceTargetNotMet;
    }
}
```
