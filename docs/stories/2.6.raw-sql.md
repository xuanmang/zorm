### 实现细节

**RawQuery 结构体定义**:
```zig
pub const RawQuery = struct {
    allocator: Allocator,
    db: *DB,
    sql: []const u8,
    args: []const QueryArg,

    pub fn exec(self: *RawQuery) !RawResult {
        return self.db.conn.execute(self.sql, self.args);
    }

    pub fn scan(self: *RawQuery, comptime T: type, dest: *std.ArrayList(T)) !void {
        const result = try self.db.conn.query(self.sql, self.args);
        defer result.deinit();

        // 复用 SelectQuery 的扫描逻辑
        try scanResult(T, result, dest, self.allocator);
    }

    pub fn scanOne(self: *RawQuery, comptime T: type) !T {
        var list: std.ArrayList(T) = .{};
        defer list.deinit(self.allocator);

        try self.scan(T, &list);

        if (list.items.len == 0) return error.NoRows;
        if (list.items.len > 1) return error.MultipleRows;

        return list.items[0];
    }

    pub fn deinit(self: *RawQuery) void {
        self.allocator.free(self.args);
    }
};
```

**参数绑定处理**:
```zig
pub fn newRaw(self: *DB, sql: []const u8, args: anytype) !RawQuery {
    const args_array = try argsToQueryArgs(self.allocator, args);

    return RawQuery{
        .allocator = self.allocator,
        .db = self,
        .sql = sql,
        .args = args_array,
    };
}

fn argsToQueryArgs(allocator: Allocator, args: anytype) ![]const QueryArg {
    const ArgsType = @TypeOf(args);
    const args_type_info = @typeInfo(ArgsType);

    if (args_type_info != .Struct) {
        @compileError("args must be a tuple");
    }

    const fields = args_type_info.Struct.fields;
    var result = try allocator.alloc(QueryArg, fields.len);

    inline for (fields, 0..) |field, i| {
        result[i] = try valueToQueryArg(@field(args, field.name));
    }

    return result;
}
```

### 使用场景示例

**窗口函数**:
```zig
const UserWithRank = struct {
    user_id: i64,
    user_name: []const u8,
    post_count: i64,
    rank: i64,
};

const sql =
    \\SELECT
    \\  u.id as user_id,
    \\  u.name as user_name,
    \\  COUNT(p.id) as post_count,
    \\  RANK() OVER (ORDER BY COUNT(p.id) DESC) as rank
    \\FROM users u
    \\LEFT JOIN posts p ON p.user_id = u.id
    \\GROUP BY u.id, u.name
    \\HAVING COUNT(p.id) > $1
    \\ORDER BY rank
    \\LIMIT $2
;

var results: std.ArrayList(UserWithRank) = .{};
defer results.deinit(allocator);

var query = try db.newRaw(sql, .{ 5, 10 });
defer query.deinit();

try query.scan(UserWithRank, &results);
```

**CTE (Common Table Expression)**:
```zig
const sql =
    \\WITH active_users AS (
    \\  SELECT id, name FROM users WHERE is_active = $1
    \\),
    \\user_stats AS (
    \\  SELECT
    \\    au.id,
    \\    au.name,
    \\    COUNT(p.id) as post_count
    \\  FROM active_users au
    \\  LEFT JOIN posts p ON p.user_id = au.id
    \\  GROUP BY au.id, au.name
    \\)
    \\SELECT * FROM user_stats WHERE post_count > $2
;

var query = try db.newRaw(sql, .{ true, 10 });
defer query.deinit();

try query.scan(UserStats, &results);
```

### 安全警告

**SQL 注入防护** [Source: architecture/coding-standards.md#禁止规则]:
```zig
// ❌ 错误 - SQL 注入风险
const user_input = "'; DROP TABLE users; --";
const sql = try std.fmt.allocPrint(allocator,
    "SELECT * FROM users WHERE name = '{s}'",
    .{user_input}
);
var query = try db.newRaw(sql, .{});

// ✅ 正确 - 使用参数绑定
const user_input = "'; DROP TABLE users; --";
const sql = "SELECT * FROM users WHERE name = $1";
var query = try db.newRaw(sql, .{user_input});
```

**文档中强调**:
- ⚠️ Raw SQL 需要手动防止 SQL 注入
- ✅ 始终使用参数绑定 ($1, $2, ...)
- ❌ 永远不要拼接用户输入到 SQL 字符串
